    /// <summary>
    /// Вычисляет значение функции y = sqrt((x-3)/(x+9))
    /// </summary>
    /// <param name="x">Аргумент функции</param>
    /// <returns>Значение функции или null, если функция не определена</returns>
    static double? CalculateFunction(double x)
    {
        // Проверяем, что подкоренное выражение неотрицательно и знаменатель не 0
        if ((x < 3) && (x >= -9))
        {
            return null; // Корень из отрицательного числа не определён в вещественных числах
        }

 

        // Возвращаем квадратный корень из дроби
        return Math.Sqrt((x - 3) / (x + 9));
    }
